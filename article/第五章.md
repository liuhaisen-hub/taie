1. 本章目标

结束时你将得到：

- 多轮记忆：同一会话内模型记住之前说过的每一句话，历史持久化在 SQLite，重启不丢；
- 会话即配置快照：会话行冻结了自己用的模型接入（key/地址/模型名），换全局配置不影响老会话；
- 协议 v2：扁平的 `delta/tool/content/error` 字段收敛为 `kind + message 消息卡`，工具调用升级为带 `meta` 的阶段状态卡片；
- 架构分层：`internal/agentkit`（运行时）/ `internal/pkg/events`（共享协议）/ `services`（传输层），eino 的可见范围被 import 图钉死在一个包里。


2. 先看架构：从"一个 agent.go"到 agentkit

第三章结束时，Agent 的全部逻辑（建模型、建 agent、搜索工具、事件循环、提示词）都挤在 internal/services/agent.go 一个文件里，internal/agent/{chat,loop,tool} 只是占位目录。本章把它们长成了真正的包：

internal/
├── agentkit/                  # Agent 运行时（唯一允许 import adk/eino 的地方）
│   ├── chatagent.go           #   ChatAgent：Runner 主流程 + 每会话模型缓存
│   ├── provider/provider.go   #   模型构建（OpenAI 兼容接入，将来多供应商适配层）
│   ├── toolkit/search.go      #   工具（web_search，沿自第三章）
│   └── workspace/             #   提示词工程（静态人设 + 动态上下文拼装）
├── pkg/
│   ├── events/                # ⭐ 两端共享的零依赖协议包（message 卡 + event 封装）
│   └── utils/                 #   小工具（TruncateWithEllipsis…）
├── services/

本章最有价值的一条架构决定：internal/pkg/events是零依赖协议包——它不 import adk、eino、gorm、wails 里的任何东西，只定义纯数据结构。Go 的依赖按包传递，如果 services 直接引用 `agentkit` 根包里的类型，编译期就会拖进整条 eino 依赖链，"传输层与运行时解耦"就只能靠自觉；而引用一个零依赖子包，安全性是 import 图机械保证的：

app ──► services ──► pkg/events ◄── agentkit（唯一 import adk/eino 的包）
                      ▲
data ─────────────────┤（实现 services.SessionRepo / agentkit.AgentRepo 接口）

于是 services/chat.go 瘦成了纯传输层，只干四件事：定位/新建会话 → 回推 begin → 调 Runner → 透传事件：

// internal/services/chat.go
type ChatAgentServices struct {
    model   ModelRepo
    session SessionRepo
    chagent *agentkit.ChatAgent
}

func (a *ChatAgentServices) ChatWithAgent(ctx context.Context, req *ChatRequest, onEvent func(events.Event) error) {
    sess, err := a.ensureSession(ctx, req)
    if err != nil {
        _ = onEvent(events.Error(fmt.Sprintf("获取会话失败: %v", err)))
        return
    }
    // Runner 启动前先把会话 ID 回推给前端：新会话此时才落库，
    // 前端收到 begin 后才能在后续请求中带上 session_id。
    if err := onEvent(events.Begin(sess.ID)); err != nil {
        return
    }
    a.chagent.Runner(ctx, sess.ID, req.UserInput, &provider.BuildParams{
        ApiKey:      sess.ApiKey,
        BaseUrl:     sess.BaseURL,
        ModalName:   sess.ModelName,
        MaxMemories: sess.MaxHistory,
    }, func(e events.Event) error {
        return onEvent(e)
    })
}

注意 `BuildParams` 的来源：**`sess.ApiKey / sess.BaseURL / sess.ModelName`**——模型接入配置不是每次实时读 `ai_model` 表，而是建会话那一刻从模型表快照进会话行。"会话冻结配置"是个刻意的产品决策：用户中途改了全局模型配置，已经聊了一半的会话不受影响（对比设计文档里"配置变更→指纹重建"的方案，快照方案把变更语义变成了'只对新会话生效'，更可预期）。代价是模型表改 Key 后老会话仍用旧 Key——本章遗留清单里有讨论。

3.1 v1 持久化粒度：为什么不落工具消息

chat_message 的列结构为工具消息（`role=tool` + `tool_call_id`）预留了位置，但 **v1 明确只写 `user` / `assistant` 两种消息**。这是设计文档里论证过的一个关键取舍，值得展开：

- 重放安全：OpenAI 兼容 API 要求 `tool` 消息必须与前面 assistant 的 `tool_calls` 消息成对出现，单独回放一条孤立的 tool 结果，部分供应商直接报错；
- 流式聚合成本：流式模式下 `tool_calls` 参数是分片到达的，落库前要按 `Index` 聚合，v1 先不做；
- 信息损失可接受：下一轮模型看不到上一轮工具返回的原文，但结论早已被消化进 assistant 回答里。

tool_calls 列先建好，v2 做全保真回放时零迁移成本。

4. 事件协议 v2：internal/pkg/events

第三章的 ChatEvent 是一堆扁平字段（`type/delta/tool/content/error`），每种事件的字段含义靠注释约定。v2 把它升级为消息卡模型：

// internal/pkg/events/response.go —— Message.Status 与 Semi AIChatDialogue 原生词表对齐
const (
    StatusLoading = "in_progress"
    StatusDone    = "completed"
    StatusError   = "failed"
)

// Message 是一张"半成品消息卡"，字段与 Semi 的 Message 一一对应
type Message struct {
    Role       string      `json:"role"`
    Content    string      `json:"content"`
    ID         string      `json:"id,omitempty"`
    CreateTime int64       `json:"createTime,omitempty"`
    Status     string      `json:"status,omitempty"`
    References []Reference `json:"references,omitempty"`
    Meta       *MessageMeta `json:"meta,omitempty"` // 思考/工具调用等中间过程
}

// MessageMeta 驱动前端渲染"阶段状态卡片"
type MessageMeta struct {
    Stage     string `json:"stage"`        // thinking / tool_call / retrieve / answer
    StageText string `json:"stage_text"`   // "调用工具 web_search"
    ToolName  string `json:"tool_name,omitempty"`
    ToolArgs  string `json:"tool_args,omitempty"`
}

// internal/pkg/events/event.go —— 事件封装 + 构造器
type Event struct {
    Kind    EventKind `json:"kind"` // begin / delta / stage / done / error
    Message Message   `json:"message"`
}

func Begin(sessionID uint64) Event { /* 会话就绪，message.id 携带会话 ID */ }
func Delta(chunk string) Event     { /* 文本增量 */ }
func Stage(meta *MessageMeta) Event{ /* 阶段变化 */ }
func Done(full string) Event       { /* 完整回答，与落库同源 */ }
func Error(text string) Event      { /* 错误 */ }

三个设计点：

1. 状态词表直接对齐 Semi。`in_progress / completed / failed` 就是 `AIChatDialogue` 的 `Message.status` 原生取值——前端收到事件几乎可以零转换地塞进气泡，连"error 状态怎么显示"都不用自己发明。
2. begin 事件解决"新会话 id 从哪来"**。`session_id=0` 发出首问时，后端建会话后第一件事就是回一条 `Begin(sess.ID)`——会话 id 放在 `message.id` 里。前端收到后记下来，后续请求全部回传。这延续了第三章"连接即会话"的模型：一次连接内的第一事件永远是身份 handshake。
3. 事件生产者分工（设计文档的契约）：`begin/error` 由 services 层构造，`delta/stage/done` 由 agentkit 产出。实际实现里 Runner 的两条错误路径也直接 `events.Error(...)` 了——功能正确，但同一语义两个生产者，改错误文案时要记得找两处。

5.3 ChatAgent.Runner：一轮生成的完整流程

ChatAgent 是注入的单例，内部 `entries map[uint64]*entry` 缓存每会话的模型实例（eino 的 ChatModel 无状态，缓存它省的是重复构建）。`Runner` 是主流程：

func (c *ChatAgent) Runner(ctx context.Context, sessionID uint64, query string,
    b *provider.BuildParams, onEvent func(events.Event) error) error {

    e, _ := c.get(ctx, sessionID, b)          // ⚠️ err 未检查，见第 9 节
    agent, _ := c.buildAgent(ctx, e, &workspace.WorkerSpace{})

    // ① 用户消息先落库——"防止其他问题中断"
    c.repo.AppendUserMessage(ctx, sessionID, query)

    // ② 读历史重放：倒序取最近 N 条，翻转回时间正序（已含刚落库的用户消息）
    message, _ := c.buildMessage(ctx, sessionID, e.maxMemories)

    // ③ 运行（流式 + 动态时间注入）
    iter := agent.Run(ctx, &adk.AgentInput{Messages: message, EnableStreaming: true},
        adk.WithSessionValues(map[string]any{"Now": time.Now().Format("2006-01-02 15:04:05")}))

    // ④ 事件循环：工具完成→Stage 卡片；流式增量→Delta；全文累积
    var full strings.Builder
    for {
        ev, ok := iter.Next()
        if !ok { break }
        if ev.Err != nil {
            if ctx.Err() != nil { // 前端断开：半截回答按"中断"落库，静默退出
                _ = c.repo.AppendAssistantMessage(ctx, sessionID, full.String(), po.MessageExitStatus)
                return ctx.Err()
            }
            return onEvent(events.Error(ev.Err.Error()))
        }
        out := ev.Output.MessageOutput
        if out == nil { continue }
        if out.Role == schema.Tool {  // 工具执行完成 → 阶段事件，前端渲染状态卡片
            return/onEvent(events.Stage(&events.MessageMeta{
                Stage: events.StageToolCall,
                StageText: fmt.Sprintf("工具 %s 执行完成", out.ToolName),
                ToolName: out.ToolName,
            })) ; continue
        }
        if out.IsStreaming {
            for {  // Recv 逐帧；空 Content 是工具调用分片，跳过
                chunk, err := out.MessageStream.Recv()
                if err == io.EOF { break }
                // ...
                full.WriteString(chunk.Content)
                if err := onEvent(events.Delta(chunk.Content)); err != nil { return err }
            }
        }
    }
    // ⑤ 完成：落库与 Done 事件同源——前端兜底全文和 DB 里的永远一致
    finishedText := full.String()
    c.repo.AppendAssistantMessage(ctx, sessionID, finishedText, po.MessageFinishStatus)
    return onEvent(events.Done(finishedText))
}

7. 前端：消费消息卡

协议变了，provider 的事件处理跟着升级，套路不变（连接生命周期、rAF 合帧都在）：

// frontend/src/hooks/aichat/provider/provider.tsx（节选）
interface ChatEvent {
    kind: "begin" | "delta" | "stage" | "done" | "error";
    message: {
        id?: string;
        content?: string;
        status?: "in_progress" | "completed" | "failed";
        meta?: ChatMeta;
    };
}

socket.onmessage = (ev: MessageEvent) => {
    const evt = ev.data as ChatEvent;
    const msg = evt.message ?? {};
    switch (evt.kind) {
        case "begin":   // 新会话此时才拿到 ID，后续请求必须回传
            sessionIdRef.current = Number(msg.id ?? 0);
            setSessionId(sessionIdRef.current);
            break;
        case "delta":
            bufRef.current += msg.content ?? "";
            scheduleFlush(botId);
            break;
        case "stage":   // 中间过程 → 写进气泡的 meta，渲染状态卡片
            setChats(prev => prev.map(m =>
                m.id === botId ? { ...m, meta: msg.meta } : m));
            break;
        case "done":    // 服务端全文兜底 + 状态收口 + 清掉阶段卡
            setChats(prev => prev.map(m =>
                m.id === botId
                    ? { ...m, content: msg.content ?? m.content, status: "completed", meta: undefined }
                    : m));
            finish();
            break;
        // error：气泡置 failed 态
    }
};

一个值得学的 React 细节：`sessionId` 同时存在于 `useState`（驱动 UI）和 `useRef`（同步副本）——因为 `sendMessage` 的 `socket.onopen` 闭包需要立即读到最新 id，不等一轮 setState 生效。
